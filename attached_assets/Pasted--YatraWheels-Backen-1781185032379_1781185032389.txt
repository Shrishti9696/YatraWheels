# ============================================================================
# YatraWheels Backend — Environment Variables
# ============================================================================
# Copy this file to `.env` and fill in the values before starting the server.
# Variables marked [REQUIRED] will cause a startup warning if missing.
# Variables marked [OPTIONAL] enable extra features but have fallbacks.
# ============================================================================

# ── Server ───────────────────────────────────────────────────────────────────
# [REQUIRED] Port the Express server listens on
PORT=5000

# [OPTIONAL] Frontend origin for CORS (defaults to http://localhost:5173)
FRONTEND_URL=http://localhost:5173

# ── Database ─────────────────────────────────────────────────────────────────
# [REQUIRED] MongoDB connection string
MONGODB_URI=mongodb://localhost:27017/yatrawheels

# ── Authentication ───────────────────────────────────────────────────────────
# [REQUIRED] Secret used to sign JWT tokens — use a long random string
JWT_SECRET=your-super-secret-jwt-key-change-this

# [OPTIONAL] JWT token expiry duration (default: 7d)
JWT_EXPIRES_IN=7d

# [OPTIONAL] Fallback session secret (used if JWT_SECRET is absent)
SESSION_SECRET=your-session-secret-change-this

# ── OpenAI (AI Travel Planner & YatraBot) ────────────────────────────────────
# [OPTIONAL] Enables AI-powered trip planning and YatraBot chat
# If missing, AI features return a "currently unavailable" response
OPENAI_API_KEY=sk-your-openai-api-key

# ── Mapbox (Live Tracking & Maps) ────────────────────────────────────────────
# [OPTIONAL] Enables live vehicle tracking on maps
# If missing, tracking features return a "currently unavailable" response
MAPBOX_TOKEN=pk.your-mapbox-public-token

# ── Razorpay (Payments) ─────────────────────────────────────────────────────
# [OPTIONAL] Enables online payment processing
# If missing, payment features return a "currently unavailable" response
RAZORPAY_KEY_ID=rzp_test_your_key_id
RAZORPAY_KEY_SECRET=your_razorpay_key_secret
RAZORPAY_WEBHOOK_SECRET=your_razorpay_webhook_secret

# ── Cloudinary (Image Uploads) ───────────────────────────────────────────────
# [OPTIONAL] Enables cloud-based image storage for vehicle photos
# If missing, images are saved locally to the /uploads folder
CLOUDINARY_CLOUD_NAME=your-cloud-name
CLOUDINARY_API_KEY=your-cloudinary-api-key
CLOUDINARY_API_SECRET=your-cloudinary-api-secret

# ── Zapier (Email Notifications) ─────────────────────────────────────────────
# [OPTIONAL] Webhook URL for sending booking confirmation emails via Zapier
ZAPIER_WEBHOOK_URL=https://hooks.zapier.com/hooks/catch/your-webhook-url
